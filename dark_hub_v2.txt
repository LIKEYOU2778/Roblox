-- ESP：原生 Highlight，不使用逐帧绘制
----------------------------------------------------------------

local espHighlights: {[Player]: Highlight} = {}
local espCharacterConnections: {[Player]: RBXScriptConnection} = {}

local function removeESP(player: Player)
	local highlight = espHighlights[player]

	if highlight then
		highlight:Destroy()
		espHighlights[player] = nil
	end
end

local function addESP(player: Player)
	if player == LocalPlayer or not localState.ESP then
		return
	end

	removeESP(player)

	local character = player.Character

	if not character then
		return
	end

	local highlight = Instance.new("Highlight")
	highlight.Name = "__CompactSandboxESP"
	highlight.Adornee = character
	highlight.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
	highlight.FillTransparency = 0.74
	highlight.OutlineTransparency = 0.08

	if player.Team then
		highlight.FillColor = player.TeamColor.Color
	else
		highlight.FillColor = COLORS.Accent2
	end

	highlight.OutlineColor = Color3.fromRGB(245, 248, 255)
	highlight.Parent = character

	espHighlights[player] = highlight
end

local function bindESPPlayer(player: Player)
	if espCharacterConnections[player] then
		espCharacterConnections[player]:Disconnect()
	end

	espCharacterConnections[player] = player.CharacterAdded:Connect(
		function()
			task.wait(0.15)

			if localState.ESP then
				addESP(player)
			end
		end
	)

	if localState.ESP then
		addESP(player)
	end
end

local function refreshESP()
	for player in pairs(espHighlights) do
		removeESP(player)
	end

	if not localState.ESP then
		return
	end

	for _, player in ipairs(Players:GetPlayers()) do
		addESP(player)
	end
end

Players.PlayerAdded:Connect(bindESPPlayer)

Players.PlayerRemoving:Connect(function(player)
	removeESP(player)

	local connection = espCharacterConnections[player]

	if connection then
		connection:Disconnect()
		espCharacterConnections[player] = nil
	end
end)

for _, player in ipairs(Players:GetPlayers()) do
	bindESPPlayer(player)
end

local lightingBaseline = {
	Brightness = Lighting.Brightness,
	ClockTime = Lighting.ClockTime,
	Ambient = Lighting.Ambient,
	OutdoorAmbient = Lighting.OutdoorAmbient,
	GlobalShadows = Lighting.GlobalShadows,
	FogStart = Lighting.FogStart,
	FogEnd = Lighting.FogEnd,
}

local atmosphereBaseline: {[Atmosphere]: {
	Density: number,
	Haze: number,
	Glare: number,
}} = {}

for _, object in ipairs(Lighting:GetDescendants()) do
	if object:IsA("Atmosphere") then
		atmosphereBaseline[object] = {
			Density = object.Density,
			Haze = object.Haze,
			Glare = object.Glare,
		}
	end
end

local originalFOVs: {[Camera]: number} = setmetatable(
	{},
	{__mode = "k"}
)

local function applyLighting()
	if localState.Fullbright then
		Lighting.Brightness = 3
		Lighting.ClockTime = 14
		Lighting.Ambient = Color3.fromRGB(190, 200, 215)
		Lighting.OutdoorAmbient = Color3.fromRGB(170, 180, 200)
		Lighting.GlobalShadows = false
	else
		Lighting.Brightness = lightingBaseline.Brightness
		Lighting.ClockTime = lightingBaseline.ClockTime
		Lighting.Ambient = lightingBaseline.Ambient
		Lighting.OutdoorAmbient = lightingBaseline.OutdoorAmbient
		Lighting.GlobalShadows = lightingBaseline.GlobalShadows
	end

	if localState.NoFog then
		Lighting.FogStart = 0
		Lighting.FogEnd = 1000000

		for atmosphere in pairs(atmosphereBaseline) do
			if atmosphere.Parent then
				atmosphere.Density = 0
				atmosphere.Haze = 0
				atmosphere.Glare = 0
			end
		end
	else
		Lighting.FogStart = lightingBaseline.FogStart
		Lighting.FogEnd = lightingBaseline.FogEnd

		for atmosphere, original in pairs(atmosphereBaseline) do
			if atmosphere.Parent then
				atmosphere.Density = original.Density
				atmosphere.Haze = original.Haze
				atmosphere.Glare = original.Glare
			end
		end
	end
end

local function applyFOV()
	local camera = Workspace.CurrentCamera

	if not camera then
		return
	end

	if originalFOVs[camera] == nil then
		originalFOVs[camera] = camera.FieldOfView
	end

	if localState.WideFOV then
		camera.FieldOfView = 100
	else
		camera.FieldOfView = originalFOVs[camera]
	end
end

----------------------------------------------------------------
-- 平滑飞行：仅开启时运行 RenderStepped
----------------------------------------------------------------

local flyConnection: RBXScriptConnection?
local flyAttachment: Attachment?
local flyVelocity: LinearVelocity?
local flyOrientation: AlignOrientation?
local flyHumanoid: Humanoid?
local flyRoot: BasePart?

local flyOriginalPlatformStand = false
local flyOriginalAutoRotate = true
local flyBoostUntil = 0

local function stopFly()
	if flyConnection then
		flyConnection:Disconnect()
		flyConnection = nil
	end

	if flyVelocity then
		flyVelocity:Destroy()
		flyVelocity = nil
	end

	if flyOrientation then
		flyOrientation:Destroy()
		flyOrientation = nil
	end

	if flyAttachment then
		flyAttachment:Destroy()
		flyAttachment = nil
	end

	if flyRoot and flyRoot.Parent then
		flyRoot.AssemblyLinearVelocity = Vector3.zero
	end

	if flyHumanoid and flyHumanoid.Parent then
		flyHumanoid.PlatformStand = flyOriginalPlatformStand
		flyHumanoid.AutoRotate = flyOriginalAutoRotate
	end

	flyHumanoid = nil
	flyRoot = nil
end

local function startFly()
	stopFly()

	if not serverState.Fly then
		return
	end

	local character = LocalPlayer.Character

	if not character then
		return
	end

	local humanoid = character:FindFirstChildOfClass("Humanoid")
	local root = character:FindFirstChild("HumanoidRootPart")

	if not humanoid or not root or not root:IsA("BasePart") then
		return
	end

	flyHumanoid = humanoid
	flyRoot = root

	flyOriginalPlatformStand = humanoid.PlatformStand
	flyOriginalAutoRotate = humanoid.AutoRotate

	humanoid.PlatformStand = true
	humanoid.AutoRotate = false

	local attachment = Instance.new("Attachment")
	attachment.Name = "__CompactFlyAttachment"
	attachment.Parent = root
	flyAttachment = attachment

	local velocity = Instance.new("LinearVelocity")
	velocity.Name = "__CompactFlyVelocity"
	velocity.Attachment0 = attachment
	velocity.RelativeTo = Enum.ActuatorRelativeTo.World
	velocity.VelocityConstraintMode =
		Enum.VelocityConstraintMode.Vector

	velocity.MaxForce = 1000000
	velocity.VectorVelocity = Vector3.zero
	velocity.Parent = root
	flyVelocity = velocity

	local orientation = Instance.new("AlignOrientation")
	orientation.Name = "__CompactFlyOrientation"
	orientation.Mode =
		Enum.OrientationAlignmentMode.OneAttachment

	orientation.Attachment0 = attachment
	orientation.MaxTorque = 1000000
	orientation.Responsiveness = 22
	orientation.RigidityEnabled = false
	orientation.Parent = root
	flyOrientation = orientation

	flyConnection = RunService.RenderStepped:Connect(function()
		if not serverState.Fly
			or not root.Parent
			or not humanoid.Parent then

			stopFly()
			return
		end

		local camera = Workspace.CurrentCamera

		if not camera then
			velocity.VectorVelocity = Vector3.zero
			return
		end

		local cameraLook = camera.CFrame.LookVector
		local cameraRight = camera.CFrame.RightVector

		local flatLook = Vector3.new(
			cameraLook.X,
			0,
			cameraLook.Z
		)

		local flatRight = Vector3.new(
			cameraRight.X,
			0,
			cameraRight.Z
		)

		if flatLook.Magnitude < 0.001 then
			flatLook = Vector3.new(
				root.CFrame.LookVector.X,
				0,
				root.CFrame.LookVector.Z
			)
		end

		if flatRight.Magnitude < 0.001 then
			flatRight = Vector3.new(1, 0, 0)
		end

		flatLook = flatLook.Unit
		flatRight = flatRight.Unit

		local moveDirection = humanoid.MoveDirection
		local forwardAmount = moveDirection:Dot(flatLook)
		local rightAmount = moveDirection:Dot(flatRight)

		-- 向前移动时，镜头俯仰控制升降
		local verticalAmount = cameraLook.Y * forwardAmount

		-- 手机原生跳跃键用于短暂向上推进
		if os.clock() < flyBoostUntil then
			verticalAmount += 1
		end

		local direction =
			flatLook * forwardAmount
			+ flatRight * rightAmount
			+ Vector3.new(0, verticalAmount, 0)

		if direction.Magnitude > 1 then
			direction = direction.Unit
		end

		velocity.VectorVelocity = direction * flySpeed

		orientation.CFrame = CFrame.lookAt(
			Vector3.zero,
			flatLook,
			Vector3.new(0, 1, 0)
		)
	end)
end

LocalPlayer.CharacterAdded:Connect(function()
	task.wait(0.3)

	if serverState.Fly then
		startFly()
	end
end)

UserInputService.JumpRequest:Connect(function()
	if serverState.Fly then
		flyBoostUntil = os.clock() + 0.22
		return
	end

	if not localState.InfiniteJump then
		return
	end

	local character = LocalPlayer.Character
	local humanoid = character
		and character:FindFirstChildOfClass("Humanoid")

	if humanoid then
		humanoid:ChangeState(
			Enum.HumanoidStateType.Jumping
		)
	end
end)

----------------------------------------------------------------
-- 手机点击传送
----------------------------------------------------------------

UserInputService.TouchTapInWorld:Connect(function(
	touchPositions,
	processedByUI
)
	if not localState.TapTP or processedByUI then
		return
	end

	local camera = Workspace.CurrentCamera
	local touchPosition = touchPositions[1]

	if not camera or not touchPosition then
		return
	end

	local ray = camera:ViewportPointToRay(
		touchPosition.X,
		touchPosition.Y
	)

	local raycastParameters = RaycastParams.new()
	raycastParameters.FilterType =
		Enum.RaycastFilterType.Exclude

	raycastParameters.FilterDescendantsInstances = {
		LocalPlayer.Character,
	}

	local result = Workspace:Raycast(
		ray.Origin,
		ray.Direction * 1000,
		raycastParameters
	)

	if result then
		ActionRemote:FireServer(
			"Teleport",
			result.Position
		)
	end
end)

local function setLocalFeature(key: string, enabled: boolean)
	if localState[key] == nil then
		return
	end

	localState[key] = enabled

	if key == "ESP" then
		refreshESP()

	elseif key == "Fullbright" or key == "NoFog" then
		applyLighting()

	elseif key == "WideFOV" then
		applyFOV()
	end

	if updateVisibleCards then
		updateVisibleCards()
	end
end

local function resetLocalFeatures()
	for key in pairs(localState) do
		localState[key] = false
	end

	refreshESP()
	applyLighting()
	applyFOV()

	if updateVisibleCards then
		updateVisibleCards()
	end
end

-- 精确计算 1/8 屏幕面积
----------------------------------------------------------------

local panelWidth = 220
local panelHeight = 190
local orbSize = 42
local edgeMargin = 8

local function getViewportSize(): Vector2
	local camera = Workspace.CurrentCamera

	if camera then
		return camera.ViewportSize
	end

	return Vector2.new(800, 600)
end

local function calculatePanelDimensions(): (number, number)
	local viewport = getViewportSize()

	-- 面积固定为整块屏幕的 1/8
	local targetArea = viewport.X * viewport.Y / 8

	-- 根据横竖屏调整宽高比，但面积不变
	local aspectRatio

	if viewport.X >= viewport.Y then
		aspectRatio = 2.05
	else
		aspectRatio = 1.15
	end

	local width = math.sqrt(targetArea * aspectRatio)
	local height = targetArea / width

	-- 理论上 1/8 面积不会超出屏幕；这里处理极端设备
	if width > viewport.X - edgeMargin * 2 then
		width = viewport.X - edgeMargin * 2
		height = targetArea / width
	end

	if height > viewport.Y - edgeMargin * 2 then
		height = viewport.Y - edgeMargin * 2
		width = targetArea / height
	end

	return math.floor(width + 0.5), math.floor(height + 0.5)
end

local function clampMainPosition(position: UDim2): UDim2
	local viewport = getViewportSize()

	local x = math.clamp(
		position.X.Offset,
		edgeMargin,
		math.max(edgeMargin, viewport.X - panelWidth - edgeMargin)
	)

	local y = math.clamp(
		position.Y.Offset,
		edgeMargin,
		math.max(edgeMargin, viewport.Y - panelHeight - edgeMargin)
	)

	return UDim2.fromOffset(x, y)
end

local function clampOrbPosition(position: UDim2): UDim2
	local viewport = getViewportSize()

	local x = math.clamp(
		position.X.Offset,
		edgeMargin,
		math.max(edgeMargin, viewport.X - orbSize - edgeMargin)
	)

	local y = math.clamp(
		position.Y.Offset,
		edgeMargin,
		math.max(edgeMargin, viewport.Y - orbSize - edgeMargin)
	)

	return UDim2.fromOffset(x, y)
end

local function updatePanelGeometry(initial: boolean?)
	panelWidth, panelHeight = calculatePanelDimensions()

	local viewport = getViewportSize()
	orbSize = math.clamp(
		math.floor(math.min(viewport.X, viewport.Y) * 0.1),
		38,
		46
	)

	Main.Size = UDim2.fromOffset(panelWidth, panelHeight)
	OpenButton.Size = UDim2.fromOffset(orbSize, orbSize)

	if initial then
		Main.Position = UDim2.fromOffset(
			viewport.X - panelWidth - edgeMargin,
			math.floor((viewport.Y - panelHeight) / 2)
		)

		OpenButton.Position = UDim2.fromOffset(
			viewport.X - orbSize - edgeMargin,
			math.floor((viewport.Y - orbSize) / 2)
		)
	else
		Main.Position = clampMainPosition(Main.Position)
		OpenButton.Position = clampOrbPosition(OpenButton.Position)
	end
end

local cards: {[string]: any} = {}

local function featureIsEnabled(definition): boolean
	if definition.mode == "serverToggle" then
		return serverState[definition.key] == true
	end

	if definition.mode == "localToggle" then
		return localState[definition.key] == true
	end

	return false
end

local function updateCard(card)
	local definition = card.definition
	local enabled = featureIsEnabled(definition)

	if definition.mode == "action" then
		card.dot.Visible = false

		if definition.danger then
			card.icon.TextColor3 = COLORS.Danger
		else
			card.icon.TextColor3 = COLORS.Accent2
		end

		return
	end

	card.dot.Visible = true

	TweenService:Create(
		card.button,
		TWEEN_FAST,
		{
			BackgroundColor3 = enabled
				and COLORS.SurfaceActive
				or COLORS.Surface,
		}
	):Play()

	TweenService:Create(
		card.stroke,
		TWEEN_FAST,
		{
			Color = enabled and COLORS.Accent or COLORS.Stroke,
			Transparency = enabled and 0.15 or 0.74,
		}
	):Play()

	TweenService:Create(
		card.dot,
		TWEEN_FAST,
		{
			BackgroundColor3 = enabled
				and COLORS.Accent
				or COLORS.TextMuted,

			BackgroundTransparency = enabled and 0 or 0.48,
		}
	):Play()

	card.icon.TextColor3 =
		enabled and COLORS.Accent or COLORS.TextMuted

	card.label.TextColor3 =
		enabled and COLORS.Text or Color3.fromRGB(205, 212, 222)
end

updateVisibleCards = function()
	for _, card in pairs(cards) do
		updateCard(card)
	end
end

	TweenService:Create(
		scale,
		TWEEN_FAST,
		{Scale = 0.95}
	):Play()

	task.delay(0.07, function()
		if scale.Parent then
			TweenService:Create(
				scale,
				TWEEN_SMOOTH,
				{Scale = 1}
			):Play()
		end
	end)
end

local function activateFeature(definition)
	if definition.mode == "serverToggle" then
		local desired = not serverState[definition.key]

		ActionRemote:FireServer(
			"Toggle",
			definition.key,
			desired
		)

	elseif definition.mode == "localToggle" then
		setLocalFeature(
			definition.key,
			not localState[definition.key]
		)

	elseif definition.mode == "action" then
		if definition.key == "Panic" then
			resetLocalFeatures()
		end

		ActionRemote:FireServer(
			"Action",
			definition.key
		)
	end
end

local function updateNavigation()
	for index, nav in ipairs(navButtons) do
		local selected = index == currentPage

		TweenService:Create(
			nav.button,
			TWEEN_FAST,
			{
				BackgroundTransparency = selected and 0.3 or 1,
				TextColor3 = selected
					and COLORS.Text
					or COLORS.TextMuted,
			}
		):Play()

		TweenService:Create(
			nav.indicator,
			TWEEN_FAST,
			{
				BackgroundTransparency = selected and 0 or 1,
			}
		):Play()
	end

local function switchPage(pageIndex: number, instant: boolean?)
	if not pages[pageIndex] then
		return
	end

	currentPage = pageIndex
	updateNavigation()

	pageSwitchSerial += 1
	local serial = pageSwitchSerial

	if instant then
		Content.GroupTransparency = 0
		Content.Position = UDim2.fromScale(0, 0.21)
		buildPage(pageIndex)
		return
	end

	TweenService:Create(
		Content,
		TWEEN_FAST,
		{
			GroupTransparency = 1,
			Position = UDim2.new(0, 8, 0.21, 0),
		}
	):Play()

		if serial ~= pageSwitchSerial then
			return
		end

		buildPage(pageIndex)

		Content.Position = UDim2.new(0, -8, 0.21, 0)

			Content,
			TWEEN_SMOOTH,
			{
				GroupTransparency = 0,
				Position = UDim2.fromScale(0, 0.21),
			}
		):Play()
	end)
end

for index, nav in ipairs(navButtons) do

local panelAnimationSerial = 0

local function positionMainFromOrb()
	local viewport = getViewportSize()
	local orbPosition = OpenButton.Position

	local orbCenterX = orbPosition.X.Offset + orbSize / 2
	local onRight = orbCenterX >= viewport.X / 2

	local targetX

	if onRight then
		targetX = viewport.X - panelWidth - edgeMargin
	else
		targetX = edgeMargin
	end

	local targetY = math.clamp(
		orbPosition.Y.Offset + orbSize / 2 - panelHeight / 2,
		edgeMargin,
		math.max(edgeMargin, viewport.Y - panelHeight - edgeMargin)
	)

	Main.Position = UDim2.fromOffset(targetX, targetY)
end

local function showPanel()
	panelAnimationSerial += 1

	positionMainFromOrb()

	OpenButton.Visible = false
	Main.Visible = true
	Main.GroupTransparency = 1
	MainScale.Scale = 0.9

	TweenService:Create(
		Main,
		TWEEN_SMOOTH,
		{GroupTransparency = 0}
	):Play()

	TweenService:Create(
		MainScale,
		TWEEN_SMOOTH,
		{Scale = 1}
	):Play()
end

local function hidePanel()
	panelAnimationSerial += 1
	local serial = panelAnimationSerial

	local viewport = getViewportSize()
	local mainCenterX = Main.Position.X.Offset + panelWidth / 2

	local orbX

	if mainCenterX < viewport.X / 2 then
		orbX = edgeMargin
	else
		orbX = viewport.X - orbSize - edgeMargin
	end

	local orbY = math.clamp(
		Main.Position.Y.Offset + panelHeight / 2 - orbSize / 2,
		edgeMargin,
		math.max(edgeMargin, viewport.Y - orbSize - edgeMargin)
	)

	OpenButton.Position = UDim2.fromOffset(orbX, orbY)

	TweenService:Create(
		Main,
		TWEEN_FAST,
		{GroupTransparency = 1}
	):Play()

	TweenService:Create(
		MainScale,
		TWEEN_FAST,
		{Scale = 0.92}
	):Play()

	task.delay(0.13, function()
		if serial ~= panelAnimationSerial then
			return
		end

		Main.Visible = false
		OpenButton.Visible = true
		OpenScale.Scale = 0.82

		TweenService:Create(
			OpenScale,
			TWEEN_SMOOTH,
			{Scale = 1}
		):Play()
	end)
end

	local delta = input.Position - orbDragStart

	if delta.Magnitude > 7 then
		orbMoved = true
	end

	OpenButton.Position = clampOrbPosition(
		UDim2.new(
			orbStartPosition.X.Scale,
			orbStartPosition.X.Offset + delta.X,
			orbStartPosition.Y.Scale,
			orbStartPosition.Y.Offset + delta.Y
		)
	)
end)

	if os.clock() - lastOrbDragEnd < 0.18 then
		return
